
import javax.swing.*;
import javax.swing.table.DefaultTableModel;
import java.awt.*;

public class StudentRegistrationGUI extends JFrame {

    JTextField nameField, rollField, emailField, phoneField;
    JComboBox<String> courseBox;
    JRadioButton male, female, other;
    JTextArea addressArea;
    DefaultTableModel model;

    StudentRegistrationGUI() {

        setTitle("Student Registration System");
        setSize(850, 650);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        JPanel mainPanel = new JPanel(new BorderLayout(10, 10));
        mainPanel.setBorder(BorderFactory.createEmptyBorder(15, 20, 15, 20));
        mainPanel.setBackground(new Color(240, 245, 250));

        JLabel heading = new JLabel("STUDENT REGISTRATION SYSTEM",
                JLabel.CENTER);
        heading.setFont(new Font("Arial", Font.BOLD, 24));
        heading.setForeground(new Color(25, 70, 120));

        mainPanel.add(heading, BorderLayout.NORTH);

        JPanel formPanel = new JPanel(new GridLayout(7, 2, 10, 12));
        formPanel.setBorder(BorderFactory.createTitledBorder(
                "Student Details"));
        formPanel.setBackground(Color.WHITE);

        nameField = new JTextField();
        rollField = new JTextField();
        emailField = new JTextField();
        phoneField = new JTextField();

        courseBox = new JComboBox<>(new String[]{
            "Information Technology",
            "Computer Engineering",
            "AI & DS",
            "Mechanical Engineering",
            "Civil Engineering"
        });

        male = new JRadioButton("Male");
        female = new JRadioButton("Female");
        other = new JRadioButton("Other");

        ButtonGroup genderGroup = new ButtonGroup();
        genderGroup.add(male);
        genderGroup.add(female);
        genderGroup.add(other);

        JPanel genderPanel = new JPanel();
        genderPanel.add(male);
        genderPanel.add(female);
        genderPanel.add(other);

        addressArea = new JTextArea(3, 20);

        formPanel.add(new JLabel("Student Name:"));
        formPanel.add(nameField);

        formPanel.add(new JLabel("Roll Number:"));
        formPanel.add(rollField);

        formPanel.add(new JLabel("Email ID:"));
        formPanel.add(emailField);

        formPanel.add(new JLabel("Phone Number:"));
        formPanel.add(phoneField);

        formPanel.add(new JLabel("Course:"));
        formPanel.add(courseBox);

        formPanel.add(new JLabel("Gender:"));
        formPanel.add(genderPanel);

        formPanel.add(new JLabel("Address:"));
        formPanel.add(new JScrollPane(addressArea));

        JPanel centerPanel = new JPanel(new BorderLayout(10, 10));
        centerPanel.setBackground(new Color(240, 245, 250));
        centerPanel.add(formPanel, BorderLayout.NORTH);

        JPanel buttonPanel = new JPanel();

        JButton registerButton = new JButton("Register");
        JButton clearButton = new JButton("Clear");
        JButton exitButton = new JButton("Exit");

        registerButton.setBackground(new Color(35, 130, 75));
        registerButton.setForeground(Color.WHITE);

        clearButton.setBackground(new Color(40, 100, 180));
        clearButton.setForeground(Color.WHITE);

        exitButton.setBackground(Color.RED);
        exitButton.setForeground(Color.WHITE);

        buttonPanel.add(registerButton);
        buttonPanel.add(clearButton);
        buttonPanel.add(exitButton);

        centerPanel.add(buttonPanel, BorderLayout.CENTER);

        String[] columns = {
            "Name", "Roll No", "Email",
            "Phone", "Course", "Gender"
        };

        model = new DefaultTableModel(columns, 0);
        JTable table = new JTable(model);
        table.setRowHeight(25);

        JScrollPane tableScroll = new JScrollPane(table);
        tableScroll.setBorder(BorderFactory.createTitledBorder(
                "Registered Students"));

        centerPanel.add(tableScroll, BorderLayout.SOUTH);

        mainPanel.add(centerPanel, BorderLayout.CENTER);

        add(mainPanel);

        registerButton.addActionListener(e -> registerStudent());
        clearButton.addActionListener(e -> clearFields());
        exitButton.addActionListener(e -> System.exit(0));

        setVisible(true);
    }

    void registerStudent() {

        String name = nameField.getText().trim();
        String roll = rollField.getText().trim();
        String email = emailField.getText().trim();
        String phone = phoneField.getText().trim();
        String course = (String) courseBox.getSelectedItem();
        String address = addressArea.getText().trim();

        String gender = "";

        if (male.isSelected()) {
            gender = "Male";
        } else if (female.isSelected()) {
            gender = "Female";
        } else if (other.isSelected()) {
            gender = "Other";
        }

        if (name.isEmpty() || roll.isEmpty()
                || email.isEmpty() || phone.isEmpty()
                || address.isEmpty() || gender.isEmpty()) {

            JOptionPane.showMessageDialog(this,
                    "Please fill all the details!");
            return;
        }

        if (!phone.matches("[0-9]{10}")) {
            JOptionPane.showMessageDialog(this,
                    "Enter a valid 10-digit phone number!");
            return;
        }

        if (!email.matches("^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+$")) {
            JOptionPane.showMessageDialog(this,
                    "Enter a valid email address!");
            return;
        }

        model.addRow(new Object[]{
            name, roll, email, phone, course, gender
        });

        JOptionPane.showMessageDialog(this,
                "Student Registered Successfully!");

        clearFields();
    }

    void clearFields() {

        nameField.setText("");
        rollField.setText("");
        emailField.setText("");
        phoneField.setText("");
        addressArea.setText("");

        male.setSelected(false);
        female.setSelected(false);
        other.setSelected(false);

        courseBox.setSelectedIndex(0);
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() ->
                new StudentRegistrationGUI());
    }
}
